    python3 q1_mobile.py ; python3 q2_fdma_tdma.py ; python3 q3_cdma.py ; python3 q3b_nearfar.py
    python3 q4_ofdma.py ; python3 q5_sdma.py ; python3 q6_decision.py ; python3 q7_redesign.py ; python3 make_figures.py
